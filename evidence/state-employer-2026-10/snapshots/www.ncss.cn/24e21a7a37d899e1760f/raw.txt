  function displaySearch () {
    $('.top-nav-load-search').click(function(){
        $(this).addClass("hidden");
        $(this).parent().find('.search-cancel').removeClass("hidden");
        $(this).parent().find('.top-nav-search-input').removeClass("hidden");

    })

  function searchBd () {
    const searchInput = document.querySelector('.search-input');
    const realQuery = document.querySelector('.realQuery');
    const searchForm = document.querySelector('.site-search');
    // 提交表单前处理搜索词
    if (searchForm) {
        searchForm.addEventListener('submit', function (e) {
            e.preventDefault(); // 阻止默认提交
            const val = searchInput.value.trim();
            if (!val) {
                alert("请输入关键词");
                return;
            }
            // 设置 realQuery 的值为关键词 + site:xxx
            realQuery.value = val + ' site:www.ncss.cn';
            // 手动提交表单
            searchForm.submit();
        });
    }
  }

            function initFn() {
                /* 切换职位 */
                $('.job-ul .job-li').click(function () {
                    var $this = $(this);
                    $('.job-ul .job-li').removeClass('selected');
                    $('.list-ul').addClass('hide');
                    $(this).addClass('selected');
                    $('.list-ul').eq($(this).index()).removeClass('hide');
                    if ($this.attr('id') == 'creer-li') {
                        $('#moreJob').attr('href', '/student/jobs/index.html?property=' + encodeURI('机关/事业单位/非营利机构'));
                    }else if ($this.attr('id') == 'gyqy-li') {
                        $('#moreJob').attr('href', '/student/jobs/index.html?property=' + encodeURI('国有企业'));
                    }else if ($this.attr('id') == 'intern-li') {
                        $('#moreJob').attr('href', '/student/jobs/internindex.html');
                    } else if ($this.attr('id') == 'key-li') {
                        $('#moreJob').attr('href', '/student/jobs/index.html?keyCode=' + encodeURI('重点领域'));
                    } else {
                        $('#moreJob').attr('href', '/student/jobs/index.html');
                    }
                });
                /* 切换站点 */
                $('.loaction-span').click(function () {
                    $('#pro-sel option').removeAttr('selected');
                    $('#pro-sel option').eq(0).attr('selected', 'selected');
                    $('#school-sel').html('');
                    $('#school-sel').select2('val', '');
                    $('#school-sel').empty();
                    if ($('#pro-sel').val() == '') {
                        $('.haschecked').each(function () {
                            if ($(this).attr('data-item') == 'pro-sel') {
                                $(this).html('');
                            }
                            if ($(this).attr('data-item') == 'school-sel') {
                                $(this).html('');
                            }
                        });
                    }
                    $('#province-sch option').removeAttr('selected');
                    $('#province-sch option').eq(0).attr('selected', 'selected');

                    $('#modal-site').modal('show');
                });
            }

            function schoollistFun(dataSfdm) {
                var innerSch = '';
                $.ajax({
                    type: 'get',
                    url: '/student/pubsite/schools',
                    cache: false,
                    data: {
                        sfdm: dataSfdm
                    },
                    success: function (data) {
                        data = escapeData(data);
                        for (var i = 0; i < data.length; i++) {
                            innerSch += '<option value="' + data[i].siteId + '">' + data[i].siteName + '</option>';
                        }
                        $('#school-sel').html(innerSch);
                        $('#school-sel').select2('val', '');
                        chooseFun();
                    }
                });
            }
            $('#school-sel').select2({
                placeholder: '请选择学校站点',
                language: 'zh-CN'
            });
            $('#province-sch').change(function () {
                schoollistFun($(this).val());
            });
            function chooseFun() {
                $('.choose').change(function () {
                    var _this = $(this);
                    if (_this.val() != null) {
                        $('.haschecked').each(function () {
                            if (_this.attr('data-item') == $(this).attr('data-item')) {
                                $(this).html('<span class="checkeditem" data-value="' + _this.val() + '">' + _this.find('option:checked').text() + '</span>');
                            }
                        });
                        siteAjaxFun();
                    }
                });
            }

            function getNews(id) {
                $.ajax({
                    type: 'get',
                    url: '/newsbyssdm',
                    cache: false,
                    data: {
                        categoryId: id,
                        ssdm: '',
                        pageSize: 20,
                        maxTitleLen: 70
                    },
                    success: function (data) {
                        $(".news-bl .load-img").addClass("hidden");
                        if (data.length > 0) {
                            //特别策划 线上id 2081267794
                            if (id != "2081267794") {
                                setxwHtml(id, data);
                            } else {
                                setztHtml(id, data);
                            }
                        } else {
                            //就业资讯 线上id 2077727632
                            if (id == '2077727632') {
                                $('.jyzx-ul').html('<p style="text-align:center;line-height:200px;font-size:16px;">暂无内容</p>');
                            }
                            //就业政策 线上id 2077745069
                            if (id == '2077745069') {
                                $('.jyzc-ul').html('<p style="text-align:center;line-height:200px;font-size:16px;">暂无内容</p>');
                            }
                            //风险提示 线上id 2135793080
                            if (id == '2135793080') {
                                $('.fxts-ul').html('<p style="text-align:center;line-height:200px;font-size:16px;">暂无内容</p>');
                            }
                            //求职经验 线上id 2077748338
                            if (id == '2077748338') {
                                $('.qzjy-ul').html('<p style="text-align:center;line-height:200px;font-size:16px;">暂无内容</p>');
                            }
                        }
                    }
                });
            }
            function setxwHtml(id, data) {
                var inner = '';
                var len = data.length > 6 ? 6 : data.length;
                for (var m = 0; m < len; m++) {
                    inner += '<li>'+ (data[m].atag ? (data[m].truncTitle) : '<a href="' + data[m].uri + '" title="' + data[m].title + '"  target="_blank">' + data[m].title + '</a>') + '</li>'
                }
                //就业资讯 线上id 2077727632
                if (id == '2077727632') {
                    $('.jyzx-ul').html(inner);
                }
                //就业政策 线上id 2077745069
                if (id == '2077745069') {
                    $('.jyzc-ul').html(inner);
                }
                //风险提示 线上id 2135793080
                if (id == '2135793080') {
                    $('.fxts-ul').html(inner);
                }
                //求职经验 线上id 2077748338
                if (id == '2077748338') {
                    $('.qzjy-ul').html(inner);
                }
            }
