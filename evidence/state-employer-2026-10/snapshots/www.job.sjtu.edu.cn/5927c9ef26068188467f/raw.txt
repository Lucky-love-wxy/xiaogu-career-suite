	function fate(time, nowTime) {
		postAjax(contextPath + 'index/common/jyrl', {date: time}, function (result) {
			var datelists = function () {
				var time = {}
				for (var i = 0; i < result.data.length; i++) {
					time[dislodgeZero(result.data[i].mark)] = ""
				}
				return time
			}
			$("#layui-laydate1").remove();
			//var currentTime = new Date(currentDateStr).getTime();
			laydate.render({
				elem: '#time' //指定元素
				, position: 'static'
				, mark: datelists()
				, isInitValue: false
				, showBottom: false
				, value: nowTime
				, ready: function (date) {
					if ((date.year + "-" + date.month + "-" + date.date) == currentDate) {
						$(".layui-laydate table tr .layui-this").attr("data-this", "thisDate")
						$(".layui-laydate table tr .layui-this").addClass("thisDate")
						
					}
					$('#time .layui-laydate-content td').each(function (index, item) {
						var ymd = $(item).attr('lay-ymd');
						if (new Date(ymd).getTime() < currentTime) {
							$(item).addClass('laydate-day-expire');
						}
					})
					$("td").mouseover(function () {
						var sd = $(this).attr("lay-ymd")
						var dataArr = result.data.find(function (value, index, arr) {
							return dislodgeZero(value.mark) === sd;
						})
						if (dataArr == undefined) {
							$(".duihua").css("display", 'none')
						} else {
							var con = [];
							for (var i = 0; i < dataArr.con.length; i++) {
								con.push("<li >");
								con.push("<span class='calendar-type calendar-type-circle");
								if ('zph' == dataArr.con[i].type) {
									con.push(' calendar-type-zph');
								}
								con.push("' style='margin-right: 10px'>");
								con.push('zph' == dataArr.con[i].type ? '招' : ('02' == dataArr.con[i].jbxs ? '空' : '宣'));
								con.push("</span>");
								con.push("<span style='margin-right: 10px'>");
								con.push(dataArr.con[i].jbsj);
								if (dataArr.con[i].zprd == '是') {
									con.push("<img src='" + contextPath + "static/index/img/red/hot.png' style='margin-left:10px;width: 32px;height: 15px;margin-top:-3px;'>");
								}
								con.push("</span>"
										, "<a href='" + contextPath + dataArr.con[i].a + "' target='_blank'>" + dataArr.con[i].con + "</a>"
										, "</li>");
							}
							$(".dataList").html(con.join(''))
							var setting = $(this).offset();
							$(".duihua").css("display", 'block')
							$(".duihua").css("position", 'absolute')
							$(".duihua").css("top", setting.top - 5)
							$(".duihua").css("left", setting.left + 55)
						}
						
					})
				}
				, change: function (value, date, endDate) {
					//切换月份时重新加载数据
					if (curentDateYear != date.year || curentDateMonth != date.month) {
						var nowTime = date.year + "-" + date.month
						var nowTimes = date.year + "-" + date.month + "-" + date.date

	function dislodgeZero(str) {
		let strArray = str.split('-');
		strArray = strArray.map(function (val) {
			if (val[0] == '0') {
				return val = val.slice(1);
			} else {
				return val;
			}
		})
		return strArray.join('-');
	}
